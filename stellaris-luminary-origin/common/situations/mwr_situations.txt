# The Last Holdouts (civil war remnants) - progress = pacification

	picture = GFX_evt_burning_settlement
	icon = "gfx/interface/icons/situation_log/situation_log_situation.dds"
	start_progress = 10
	monthly_progress = {
		base = 0.5
		modifier = { add = 0.5 owner = { has_country_flag = mwr_holdouts_crush } }
		modifier = { add = 0.25 owner = { has_country_flag = mwr_holdouts_reconcile } }
	}
	approaches = {
		situation_mwr_holdouts_crush = {
			icon = "gfx/interface/icons/situation_log/situation_log_situation.dds"

			modifier = { country_unity_produces_mult = -0.05 }
		}
		situation_mwr_holdouts_reconcile = {
			icon = "gfx/interface/icons/situation_log/situation_log_situation.dds"
			on_select = { owner = { set_country_flag = mwr_holdouts_reconcile remove_country_flag = mwr_holdouts_crush } }
			modifier = { country_influence_produces_add = -1 }
		}
	}
	stages = {
		situation_mwr_holdouts_1 = { end = 40 icon = "gfx/interface/icons/situation_log/situation_log_situation.dds"
			modifier = { planet_stability_add = -10 planet_crime_add = 10 } }
		situation_mwr_holdouts_2 = { end = 80 icon = "gfx/interface/icons/situation_log/situation_log_situation.dds"
			on_first_enter = { owner = { country_event = { id = mwr_holdouts.2 } } }
			modifier = { planet_stability_add = -5 } }
		situation_mwr_holdouts_3 = { end = 100 icon = "gfx/interface/icons/situation_log/situation_log_situation.dds"
			on_first_enter = { owner = { country_event = { id = mwr_holdouts.3 } } } }
	}
	on_progress_complete = { owner = { country_event = { id = mwr_holdouts.4 } } }
}
