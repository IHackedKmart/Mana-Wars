############ THE UNIFYING PROMISE (checked yearly) ############
namespace = mwr_promise

country_event = { id = mwr_promise.100 hide_window = yes is_triggered_only = yes
	trigger = { has_country_flag = mwr_promises_active }
	immediate = {
		change_variable = { which = mwr_promise_year value = 1 }
		# Promise 1: Leave the ashes - 3 worlds within 20 years
		if = { limit = { NOT = { has_country_flag = mwr_p1_done } num_owned_planets >= 3 }
			set_country_flag = mwr_p1_done country_event = { id = mwr_promise.1 } }
		else_if = { limit = { NOT = { has_country_flag = mwr_p1_done } check_variable = { which = mwr_promise_year value >= 20 } }
			set_country_flag = mwr_p1_done set_country_flag = mwr_p_broken country_event = { id = mwr_promise.11 } }
		# Promise 2: Never again - Psionic Theory within 30 years
		if = { limit = { NOT = { has_country_flag = mwr_p2_done } has_technology = tech_psionic_theory }
			set_country_flag = mwr_p2_done country_event = { id = mwr_promise.2 } }
		else_if = { limit = { NOT = { has_country_flag = mwr_p2_done } check_variable = { which = mwr_promise_year value >= 30 } }
			set_country_flag = mwr_p2_done set_country_flag = mwr_p_broken country_event = { id = mwr_promise.12 } }
		# Promise 3: Rebuild - Healthy capital (heal the tomb world) within 40 years
		if = { limit = { NOT = { has_country_flag = mwr_p3_done } capital_scope = { NOT = { has_modifier = mwr_whispering_scars } } num_owned_planets >= 6 }
			set_country_flag = mwr_p3_done country_event = { id = mwr_promise.3 } }
		else_if = { limit = { NOT = { has_country_flag = mwr_p3_done } check_variable = { which = mwr_promise_year value >= 40 } }
			set_country_flag = mwr_p3_done set_country_flag = mwr_p_broken country_event = { id = mwr_promise.13 } }
		if = { limit = { has_country_flag = mwr_p1_done has_country_flag = mwr_p2_done has_country_flag = mwr_p3_done }
			remove_country_flag = mwr_promises_active country_event = { id = mwr_promise.20 } }
	}
}
