namespace = mwr_luminary

# Setup: nuked capital, the Luminary takes the throne
country_event = {
	id = mwr_luminary.1
	hide_window = yes
	is_triggered_only = yes
	trigger = { has_origin = origin_mwr_luminary }
	immediate = {
		capital_scope = {
			change_pc = pc_nuked
			add_modifier = { modifier = mwr_mindward_bastion days = -1 }
			add_modifier = { modifier = mwr_whispering_scars days = -1 }
			add_deposit = d_ruined_cities
			add_deposit = d_ruined_cities
		}
		add_modifier = { modifier = mwr_oath_of_stars days = -1 }
		ruler = {
			add_trait = leader_trait_mwr_luminary
			set_immortal = yes
			set_leader_flag = mwr_the_luminary
		}
		country_event = { id = mwr_prologue.1 days = 1 }
		start_situation = { type = situation_mwr_holdouts target = capital_scope }
		start_situation = { type = situation_mwr_whisperers target = capital_scope }
		country_event = { id = mwr_vaults.1 days = 1800 random = 720 }
		country_event = { id = mwr_luminary.3 days = 3600 random = 720 }
	}
}
